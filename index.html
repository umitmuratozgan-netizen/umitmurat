1 · Kurulum

pip install chatterbox-tts

2 · Konuşma üret

from chatterbox.tts_turbo import ChatterboxTurboTTS
import torchaudio

model = ChatterboxTurboTTS.from_pretrained(device="cuda")

# Paralinguistik etiketlerle temel üretim
wav = model.generate(
    "I can't believe it [laugh] that's hilarious!",
    audio_prompt_path="reference.wav",
)

torchaudio.save("output.wav", wav, model.sr)
from chatterbox.mtl_tts import ChatterboxMultilingualTTS
import torchaudio

model = ChatterboxMultilingualTTS.from_pretrained(device="cuda")

# Ses klonlama ile Fransızca konuşma üret
wav = model.generate(
    "Bonjour, comment allez-vous aujourd'hui?",
    audio_prompt_path="reference.wav",
    language_id="fr",
)

torchaudio.save("output_fr.wav", wav, model.sr)
from chatterbox.tts import ChatterboxTTS
import torchaudio

model = ChatterboxTTS.from_pretrained(device="cuda")

# Dışavurumlu seslendirme için abartmayı kullan
wav = model.generate(
    "The storm raged on through the darkest night.",
    audio_prompt_path="reference.wav",
    exaggeration=0.7,
    cfg_weight=0.5,
)

torchaudio.save("output.wav", wav, model.sr)
import perth, librosa

# Chatterbox ile üretilen seslerdeki yapay zeka filigranını tespit et
audio, sr = librosa.load("output.wav", sr=None)

watermarker = perth.PerthImplicitWatermarker()
result = watermarker.get_watermark(audio, sample_rate=sr)

print(f"Filigran tespit edildi: {result}")
